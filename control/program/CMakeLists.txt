cmake_minimum_required(VERSION 3.16)
project(poke_chassis_lift LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
add_library(elmue_can STATIC
 third_party/elmue_can/src/socketcan.cpp
 third_party/elmue_can/src/elmue_can.cpp
 third_party/elmue_can/src/robstride_protocol.cpp
 third_party/elmue_can/src/robstride_elmue.cpp
 third_party/elmue_can/src/zeroerr_canopen.cpp)
target_include_directories(elmue_can PUBLIC third_party/elmue_can/include)
target_link_libraries(elmue_can PUBLIC Threads::Threads PRIVATE PkgConfig::LIBUSB)
file(GLOB RUCKIG_SOURCES CONFIGURE_DEPENDS third_party/ruckig/src/ruckig/*.cpp)
list(FILTER RUCKIG_SOURCES EXCLUDE REGEX "(cloud_client|python).cpp$")
add_library(ruckig STATIC ${RUCKIG_SOURCES})
target_include_directories(ruckig PUBLIC third_party/ruckig/include)
add_library(lift_profile SHARED remote/lift_profile.cpp)
target_link_libraries(lift_profile PRIVATE ruckig)
add_library(poke_chassis_lift STATIC src/auxiliary.cpp src/chassis_xdrive_odom.cpp src/base.cpp)
target_include_directories(poke_chassis_lift PUBLIC include PRIVATE src)
target_link_libraries(poke_chassis_lift PUBLIC elmue_can Threads::Threads PRIVATE ruckig)
target_compile_options(poke_chassis_lift PRIVATE -Wall -Wextra -Wpedantic)
option(BUILD_CONTROL_TESTS "Build original offline tests" ON)
if(BUILD_CONTROL_TESTS)
 enable_testing()
 foreach(name base chassis auxiliary_config)
  add_executable(${name}_tests tests/${name}_tests.cpp)
  target_include_directories(${name}_tests PRIVATE src)
  target_link_libraries(${name}_tests PRIVATE poke_chassis_lift)
  add_test(NAME ${name}_tests COMMAND ${name}_tests)
 endforeach()
endif()
add_executable(config_reference examples/config_reference.cpp)
target_link_libraries(config_reference PRIVATE poke_chassis_lift)

add_executable(remote_bridge remote/bridge.cpp)
target_include_directories(remote_bridge PRIVATE src)
target_link_libraries(remote_bridge PRIVATE poke_chassis_lift)
target_compile_options(remote_bridge PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_CONTROL_TESTS)
 add_executable(socketcan_tests tests/socketcan_tests.cpp)
 target_link_libraries(socketcan_tests PRIVATE elmue_can)
 add_test(NAME socketcan_tests COMMAND socketcan_tests)
endif()
add_executable(check_can remote/check_can.cpp)
target_link_libraries(check_can PRIVATE elmue_can)
if(BUILD_CONTROL_TESTS)
 add_executable(wheel_balance_tests tests/wheel_balance_tests.cpp)
 target_include_directories(wheel_balance_tests PRIVATE remote)
 add_test(NAME wheel_balance_tests COMMAND wheel_balance_tests)
endif()
if(BUILD_CONTROL_TESTS)
 add_executable(body_control_tests tests/body_control_tests.cpp)
 target_include_directories(body_control_tests PRIVATE remote include)
 add_test(NAME body_control_tests COMMAND body_control_tests)
endif()

add_executable(single_wheel remote/single_wheel.cpp)
target_include_directories(single_wheel PRIVATE src)
target_link_libraries(single_wheel PRIVATE poke_chassis_lift)
target_compile_options(single_wheel PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_CONTROL_TESTS)
 add_executable(feedback_snapshot_tests tests/feedback_snapshot_tests.cpp)
 target_link_libraries(feedback_snapshot_tests PRIVATE elmue_can)
 add_test(NAME feedback_snapshot_tests COMMAND feedback_snapshot_tests)
endif()
